测试 2: 从 YouTube 页面提取

需要在 YouTube 视频页面的控制台运行

// 在 YouTube 视频页面控制台运行：
const response = window.ytInitialPlayerResponse;
console.log('ytInitialPlayerResponse:', response);
const tracks = response?.captions?.playerCaptionsTracklistRenderer?.captionTracks;
console.log('Caption tracks:', tracks);
if (tracks && tracks.length > 0) {
  console.log('First track baseUrl:', tracks[0].baseUrl);
  // 测试下载
  fetch(tracks[0].baseUrl + '&fmt=json3')
    .then(r => r.text())
    .then(text => console.log('Subtitle data:', text.substring(0, 500)));
}
