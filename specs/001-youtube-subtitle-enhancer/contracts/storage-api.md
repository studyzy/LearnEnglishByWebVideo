# Chrome Storage API合同

**版本**: 1.0.0  
**日期**: 2026-01-20

## 概述

本文档定义Chrome扩展使用`chrome.storage.local` API的数据存储规范，包括所有存储键、数据结构和访问方法。

### 1.1 命名约定

```typescript

// 视频缓存
const VIDEO_CACHE_KEY_PREFIX = 'video_cache_'; // + videoId

// 统计数据
```

### 2.1 UserProfile (用户配置)

**Storage Key**: `user_profile`

**TypeScript Interface**:
```typescript
interface UserProfile {
  // 英语水平
  englishLevel: 'beginner' | 'intermediate' | 'advanced';
  
  // LLM配置
  llmProvider: 'openai' | 'claude';
  llmApiKey: string;
  llmModel?: string;
  
  // 词汇表
  masteredWords: Array<{
    word: string;
    lemma?: string;
    addedTime: number;
    source?: 'manual' | 'interactive';
  }>;
  focusWords: Array<{
    word: string;
    lemma?: string;
    addedTime: number;
    source?: 'manual' | 'interactive';
  }>;
  
  // UI偏好
  subtitleFontSize: number;
  subtitleFontColor: string;
  subtitleBackgroundColor: string;
  translationColor: string;
  
  // 扩展状态
  isEnabled: boolean;
  showOnboardingGuide: boolean;
  
  // 元数据
  createdAt: number;
  lastUsedAt: number;
  totalVideosWatched?: number;
  totalWordsLearned?: number;
}
```

**默认值**:
```typescript
const DEFAULT_USER_PROFILE: UserProfile = {
  englishLevel: 'intermediate',
  llmProvider: 'openai',
  llmApiKey: '',
  masteredWords: [],
  focusWords: [],
  subtitleFontSize: 16,
  subtitleFontColor: '#ffffff',
  subtitleBackgroundColor: 'rgba(0,0,0,0.8)',
  translationColor: '#ffeb3b',
  isEnabled: true,
  showOnboardingGuide: true,
  createdAt: Date.now(),
  lastUsedAt: Date.now(),
};
```

**API方法**:
```typescript
/**
 * 读取用户配置
 */
async function getUserProfile(): Promise<UserProfile> {
  const result = await chrome.storage.local.get('user_profile');
  return result.user_profile || DEFAULT_USER_PROFILE;
}

/**
 * 更新用户配置(部分更新)
 */
async function updateUserProfile(updates: Partial<UserProfile>): Promise<UserProfile> {
  const current = await getUserProfile();
  const updated = { ...current, ...updates, lastUsedAt: Date.now() };
  await chrome.storage.local.set({ user_profile: updated });
  return updated;
}

/**
 * 重置用户配置为默认值
 */
async function resetUserProfile(): Promise<void> {
  const resetProfile = {
    ...DEFAULT_USER_PROFILE,
    createdAt: Date.now(),
    lastUsedAt: Date.now(),
  };
  await chrome.storage.local.set({ user_profile: resetProfile });
}
```

### 2.2 VideoSession (视频缓存)

**Storage Key**: `video_cache_{videoId}`  
示例: `video_cache_dQw4w9WgXcQ`

**TypeScript Interface**:
```typescript
interface VideoSession {
  videoId: string;
  videoTitle?: string;
  
  // 字幕元数据
  subtitleLanguage: string;
  subtitleTrackUrl: string;
  isAutoGenerated: boolean;
  
  // 字幕内容
  segments: Array<{
    startTime: number;
    endTime: number;
    originalText: string;
    enhancedHTML: string;
    words: Array<{
      word: string;
      lemma: string;
      isUnknown: boolean;
      translation?: string;
      position: { start: number; end: number };
    }>;
  }>;
  
  // 处理状态
  isProcessing: boolean;
  processingProgress: number;
  lastProcessedTime: number;
  
  // 缓存元数据
  cacheExpiry: number;
}
```

**API方法**:
```typescript
/**
 * 缓存视频会话
 * @param session 视频会话数据
 */
async function cacheVideoSession(session: VideoSession): Promise<void> {
  const cacheKey = `video_cache_${session.videoId}`;
  const cacheData = {
    ...session,
    cacheExpiry: Date.now() + 7 * 24 * 60 * 60 * 1000, // 7天
  };
  await chrome.storage.local.set({ [cacheKey]: cacheData });
}

/**
 * 获取缓存的视频会话
 * @param videoId YouTube视频ID
 * @returns 缓存的会话数据，如果不存在或已过期则返回null
 */
async function getCachedVideoSession(videoId: string): Promise<VideoSession | null> {
  const cacheKey = `video_cache_${videoId}`;
  const result = await chrome.storage.local.get(cacheKey);
  const cached = result[cacheKey];
  
  if (!cached) return null;
  
  // 检查过期
  if (Date.now() > cached.cacheExpiry) {
    await chrome.storage.local.remove(cacheKey);
    return null;
  }
  
  return cached;
}

/**
 * 删除视频缓存
 */
async function deleteVideoCache(videoId: string): Promise<void> {
  const cacheKey = `video_cache_${videoId}`;
  await chrome.storage.local.remove(cacheKey);
}

/**
 * 清理所有过期的视频缓存
 */
async function cleanupExpiredVideoCaches(): Promise<number> {
  const allData = await chrome.storage.local.get(null);
  const keysToRemove: string[] = [];
  
  for (const [key, value] of Object.entries(allData)) {
    if (key.startsWith('video_cache_')) {
      const session = value as VideoSession;
      if (Date.now() > session.cacheExpiry) {
        keysToRemove.push(key);
      }
    }
  }
  
  if (keysToRemove.length > 0) {
    await chrome.storage.local.remove(keysToRemove);
  }
  
  return keysToRemove.length;
}
```

### 2.3 TranslationCache (翻译缓存)

**Storage Key**: `translation_cache_{contextHash}`  
示例: `translation_cache_run_2k8x9a`

**TypeScript Interface**:
```typescript
interface TranslationCacheEntry {
  word: string;
  context: string; // 上下文(前50字符)
  contextHash: string;
  translation: string;
  timestamp: number;
  expiryTime: number;
  llmProvider: 'openai' | 'claude';
  llmModel: string;
}
```

**API方法**:
```typescript
/**
 * 生成缓存键
 * @param word 单词
 * @param context 完整上下文
 * @returns 缓存键
 */
function generateTranslationCacheKey(word: string, context: string): string {
  const snippet = context.slice(0, 50);
  const hash = simpleHash(`${word}_${snippet}`);
  return `translation_cache_${hash}`;
}

/**
 * 缓存翻译结果
 */
async function cacheTranslation(
  word: string,
  context: string,
  translation: string,
  provider: 'openai' | 'claude',
  model: string
): Promise<void> {
  const cacheKey = generateTranslationCacheKey(word, context);
  const entry: TranslationCacheEntry = {
    word,
    context: context.slice(0, 50),
    contextHash: cacheKey.replace('translation_cache_', ''),
    translation,
    timestamp: Date.now(),
    expiryTime: Date.now() + 7 * 24 * 60 * 60 * 1000, // 7天
    llmProvider: provider,
    llmModel: model,
  };
  await chrome.storage.local.set({ [cacheKey]: entry });
}

/**
 * 获取缓存的翻译
 */
async function getTranslationCache(
  word: string,
  context: string
): Promise<string | null> {
  const cacheKey = generateTranslationCacheKey(word, context);
  const result = await chrome.storage.local.get(cacheKey);
  const entry = result[cacheKey] as TranslationCacheEntry;
  
  if (!entry) return null;
  
  // 检查过期
  if (Date.now() > entry.expiryTime) {
    await chrome.storage.local.remove(cacheKey);
    return null;
  }
  
  return entry.translation;
}

/**
 * 清理过期的翻译缓存
 */
async function cleanupExpiredTranslationCaches(): Promise<number> {
  const allData = await chrome.storage.local.get(null);
  const keysToRemove: string[] = [];
  
  for (const [key, value] of Object.entries(allData)) {
    if (key.startsWith('translation_cache_')) {
      const entry = value as TranslationCacheEntry;
      if (Date.now() > entry.expiryTime) {
        keysToRemove.push(key);
      }
    }
  }
  
  if (keysToRemove.length > 0) {
    await chrome.storage.local.remove(keysToRemove);
  }
  
  return keysToRemove.length;
}

/**
 * 简单哈希函数(用于生成缓存键)
 */
function simpleHash(str: string): string {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    const char = str.charCodeAt(i);
    hash = (hash << 5) - hash + char;
    hash = hash & hash;
  }
  return Math.abs(hash).toString(36);
}
```

## 3. 存储空间管理

### 3.1 配额监控

```typescript
/**
 * 获取存储使用情况
 */
async function getStorageUsage(): Promise<{
  bytesInUse: number;
  bytesLimit: number;
  percentUsed: number;
}> {
  // chrome.storage.local配额: ~5-10 MB
  const QUOTA_BYTES = 10 * 1024 * 1024; // 10 MB
  
  // 获取所有数据
  const allData = await chrome.storage.local.get(null);
  const dataString = JSON.stringify(allData);
  const bytesInUse = new Blob([dataString]).size;
  
  return {
    bytesInUse,
    bytesLimit: QUOTA_BYTES,
    percentUsed: (bytesInUse / QUOTA_BYTES) * 100,
  };
}

/**
 * 检查存储配额并触发清理
 */
async function checkStorageQuota(): Promise<void> {
  const usage = await getStorageUsage();
  
  if (usage.percentUsed > 80) {
    console.warn(`Storage usage at ${usage.percentUsed.toFixed(2)}%, triggering cleanup`);
    await cleanupOldestCaches();
  }
}

/**
 * 清理最老的缓存(30%)
 */
async function cleanupOldestCaches(): Promise<void> {
  const allData = await chrome.storage.local.get(null);
  const caches: Array<{ key: string; timestamp: number }> = [];
  
  // 收集所有缓存
  for (const [key, value] of Object.entries(allData)) {
    if (key.startsWith('video_cache_')) {
      caches.push({ key, timestamp: (value as VideoSession).lastProcessedTime });
    } else if (key.startsWith('translation_cache_')) {
      caches.push({ key, timestamp: (value as TranslationCacheEntry).timestamp });
    }
  }
  
  // 按时间排序，删除最老的30%
  caches.sort((a, b) => a.timestamp - b.timestamp);
  const toDelete = caches.slice(0, Math.floor(caches.length * 0.3));
  
  await chrome.storage.local.remove(toDelete.map(c => c.key));
  console.log(`Cleaned up ${toDelete.length} cache entries`);
}
```

## 4. 批量操作优化

### 4.1 批量读取

❌ **低效: 多次单独读取**
```typescript
const profile = await chrome.storage.local.get('user_profile');
const stats = await chrome.storage.local.get('statistics');
const version = await chrome.storage.local.get('data_version');
```

✅ **高效: 一次批量读取**
```typescript
const result = await chrome.storage.local.get([
  'user_profile',
  'statistics',
  'data_version',
]);
const profile = result.user_profile;
const stats = result.statistics;
const version = result.data_version;
```

### 4.2 批量写入

❌ **低效: 多次单独写入**
```typescript
await chrome.storage.local.set({ user_profile: profile });
await chrome.storage.local.set({ statistics: stats });
await chrome.storage.local.set({ data_version: version });
```

✅ **高效: 一次批量写入**
```typescript
await chrome.storage.local.set({
  user_profile: profile,
  statistics: stats,
  data_version: version,
});
```

## 8. 最佳实践

### 8.1 错误处理

```typescript
/**
 * 安全的存储读取(带错误处理)
 */
async function safeGet<T>(key: string, defaultValue: T): Promise<T> {
  try {
    const result = await chrome.storage.local.get(key);
    return result[key] !== undefined ? result[key] : defaultValue;
  } catch (error) {
    console.error(`Failed to read from storage: ${key}`, error);
    return defaultValue;
  }
}

/**
 * 安全的存储写入(带错误处理)
 */
async function safeSet(data: Record<string, any>): Promise<boolean> {
  try {
    await chrome.storage.local.set(data);
    return true;
  } catch (error) {
    console.error('Failed to write to storage', error);
    if (error.message?.includes('QUOTA_BYTES')) {
      // 存储配额超限，触发清理
      await cleanupOldestCaches();
    }
    return false;
  }
}
```

### 8.2 性能优化

1. **缓存常用数据**: 在内存中缓存`user_profile`，减少chrome.storage读取
2. **延迟写入**: 使用debounce延迟非关键数据写入
3. **批量操作**: 尽可能使用批量读写API
4. **定期清理**: 每天运行一次过期缓存清理
